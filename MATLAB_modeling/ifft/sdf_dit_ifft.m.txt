function y = sdf_dit_ifft(x, T) %#codegen

N  = 4096;
sc = 1/N;
% ifft(x) = swap(fft(swap(x))) / N, with swap exchanging the real and imaginary parts
% dit core: bit-reversed order in, natural order out
% 1/N = 2^-12, applied at the output as a binary point move (no bits lost)

% flip I and Q at the input (no shift, the input is already at the level the stages were sized for)

% twiddle ROM in hardware
idxVec         = (0:2047).';
twTable_double = exp(+1i * 2 * pi * idxVec / N);   % double
twTable        = cast(twTable_double, 'like', T.WN);   % casting




%stage 1: period=2, half=1 (twiddle always 1 -- no multiply/cast needed)
stg_1 = cast(complex(zeros(4096,1)), 'like', T.stg1);
for i = 1:4096
    p = mod(i-1, 2);
    if p < 1
        stg_1(i) = cast(x(i) + x(i + 1), 'like', T.stg1);
    else
        stg_1(i) = cast(x(i - 1) - x(i), 'like', T.stg1);
    end
end

%stage 2: period=4, half=2, exponent step = 2^10 = 1024
stg_2 = cast(complex(zeros(4096,1)), 'like', T.stg2);
for i = 1:4096
    p = mod(i-1, 4);
    if p < 2
        n  = p;
        Wp = twTable(1024*n + 1);      % WN^(1024*n)
        stg_2(i) = cast(stg_1(i) + stg_1(i + 2) * Wp, 'like', T.stg2);
    else
        n  = p - 2;
        Wp = twTable(1024*n + 1);      % WN^(1024*n)
        stg_2(i) = cast(stg_1(i - 2) - stg_1(i) * Wp, 'like', T.stg2);
    end
end

%stage 3: period=8, half=4, exponent step = 2^9 = 512
stg_3 = cast(complex(zeros(4096,1)), 'like', T.stg3);
for i = 1:4096
    p = mod(i-1, 8);
    if p < 4
        n  = p;
        Wp = twTable(512*n + 1);       % WN^(512*n)
        stg_3(i) = cast(stg_2(i) + stg_2(i + 4) * Wp, 'like', T.stg3);
    else
        n  = p - 4;
        Wp = twTable(512*n + 1);       % WN^(512*n)
        stg_3(i) = cast(stg_2(i - 4) - stg_2(i) * Wp, 'like', T.stg3);
    end
end

%stage 4: period=16, half=8, exponent step = 2^8 = 256
stg_4 = cast(complex(zeros(4096,1)), 'like', T.stg4);
for i = 1:4096
    p = mod(i-1, 16);
    if p < 8
        n  = p;
        Wp = twTable(256*n + 1);       % WN^(256*n)
        stg_4(i) = cast(stg_3(i) + stg_3(i + 8) * Wp, 'like', T.stg4);
    else
        n  = p - 8;
        Wp = twTable(256*n + 1);       % WN^(256*n)
        stg_4(i) = cast(stg_3(i - 8) - stg_3(i) * Wp, 'like', T.stg4);
    end
end

%stage 5: period=32, half=16, exponent step = 2^7 = 128
stg_5 = cast(complex(zeros(4096,1)), 'like', T.stg5);
for i = 1:4096
    p = mod(i-1, 32);
    if p < 16
        n  = p;
        Wp = twTable(128*n + 1);       % WN^(128*n)
        stg_5(i) = cast(stg_4(i) + stg_4(i + 16) * Wp, 'like', T.stg5);
    else
        n  = p - 16;
        Wp = twTable(128*n + 1);       % WN^(128*n)
        stg_5(i) = cast(stg_4(i - 16) - stg_4(i) * Wp, 'like', T.stg5);
    end
end

%stage 6: period=64, half=32, exponent step = 2^6 = 64
stg_6 = cast(complex(zeros(4096,1)), 'like', T.stg6);
for i = 1:4096
    p = mod(i-1, 64);
    if p < 32
        n  = p;
        Wp = twTable(64*n + 1);        % WN^(64*n)
        stg_6(i) = cast(stg_5(i) + stg_5(i + 32) * Wp, 'like', T.stg6);
    else
        n  = p - 32;
        Wp = twTable(64*n + 1);        % WN^(64*n)
        stg_6(i) = cast(stg_5(i - 32) - stg_5(i) * Wp, 'like', T.stg6);
    end
end

%stage 7: period=128, half=64, exponent step = 2^5 = 32
stg_7 = cast(complex(zeros(4096,1)), 'like', T.stg7);
for i = 1:4096
    p = mod(i-1, 128);
    if p < 64
        n  = p;
        Wp = twTable(32*n + 1);        % WN^(32*n)
        stg_7(i) = cast(stg_6(i) + stg_6(i + 64) * Wp, 'like', T.stg7);
    else
        n  = p - 64;
        Wp = twTable(32*n + 1);        % WN^(32*n)
        stg_7(i) = cast(stg_6(i - 64) - stg_6(i) * Wp, 'like', T.stg7);
    end
end

%stage 8: period=256, half=128, exponent step = 2^4 = 16
stg_8 = cast(complex(zeros(4096,1)), 'like', T.stg8);
for i = 1:4096
    p = mod(i-1, 256);
    if p < 128
        n  = p;
        Wp = twTable(16*n + 1);        % WN^(16*n)
        stg_8(i) = cast(stg_7(i) + stg_7(i + 128) * Wp, 'like', T.stg8);
    else
        n  = p - 128;
        Wp = twTable(16*n + 1);        % WN^(16*n)
        stg_8(i) = cast(stg_7(i - 128) - stg_7(i) * Wp, 'like', T.stg8);
    end
end

%stage 9: period=512, half=256, exponent step = 2^3 = 8
stg_9 = cast(complex(zeros(4096,1)), 'like', T.stg9);
for i = 1:4096
    p = mod(i-1, 512);
    if p < 256
        n  = p;
        Wp = twTable(8*n + 1);         % WN^(8*n)
        stg_9(i) = cast(stg_8(i) + stg_8(i + 256) * Wp, 'like', T.stg9);
    else
        n  = p - 256;
        Wp = twTable(8*n + 1);         % WN^(8*n)
        stg_9(i) = cast(stg_8(i - 256) - stg_8(i) * Wp, 'like', T.stg9);
    end
end

%stage 10: period=1024, half=512, exponent step = 2^2 = 4
stg_10 = cast(complex(zeros(4096,1)), 'like', T.stg10);
for i = 1:4096
    p = mod(i-1, 1024);
    if p < 512
        n  = p;
        Wp = twTable(4*n + 1);         % WN^(4*n)
        stg_10(i) = cast(stg_9(i) + stg_9(i + 512) * Wp, 'like', T.stg10);
    else
        n  = p - 512;
        Wp = twTable(4*n + 1);         % WN^(4*n)
        stg_10(i) = cast(stg_9(i - 512) - stg_9(i) * Wp, 'like', T.stg10);
    end
end

%stage 11: period=2048, half=1024, exponent step = 2^1 = 2
stg_11 = cast(complex(zeros(4096,1)), 'like', T.stg11);
for i = 1:4096
    p = mod(i-1, 2048);
    if p < 1024
        n  = p;
        Wp = twTable(2*n + 1);         % WN^(2*n)
        stg_11(i) = cast(stg_10(i) + stg_10(i + 1024) * Wp, 'like', T.stg11);
    else
        n  = p - 1024;
        Wp = twTable(2*n + 1);         % WN^(2*n)
        stg_11(i) = cast(stg_10(i - 1024) - stg_10(i) * Wp, 'like', T.stg11);
    end
end

%stage 12: period=4096, half=2048, exponent step = 2^0 = 1
stg_12 = cast(complex(zeros(4096,1)), 'like', T.stg12);
for i = 1:4096
    if i <= 2048
        n  = i - 1;
        Wp = twTable(n + 1);           % WN^(1*n)
        stg_12(i) = cast(stg_11(i) + stg_11(i + 2048) * Wp, 'like', T.stg12);
    else
        n  = i - 2049;
        Wp = twTable(n + 1);           % WN^(1*n)
        stg_12(i) = cast(stg_11(i - 2048) - stg_11(i) * Wp, 'like', T.stg12);
    end
end

%ntOut = numerictype(1, T.y.WordLength, T.y.FractionLength + 12);
%yc = reinterpretcast(stg_12,ntOut);
%y =cast(yc, 'like', T.x);
y = cast(stg_12 .* sc, 'like', T.y);
end
